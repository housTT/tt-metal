#!/usr/bin/env python3
"""Generate circuit for fib(isqrt(N)) % 2^32"""

class Circuit:
    def __init__(self):
        self.gates = []
        self.next_sig = 32  # signals 0-31 are inputs
    
    def new(self):
        s = self.next_sig
        self.next_sig += 1
        return s
    
    def copy(self, out, inp):
        self.gates.append(f"out{out} = out{inp}")
        return out
    
    def not_g(self, inp):
        s = self.new()
        self.gates.append(f"out{s} = ~out{inp}")
        return s
    
    def and_g(self, inp1, inp2):
        s = self.new()
        self.gates.append(f"out{s} = out{inp1} & out{inp2}")
        return s
    
    def or_g(self, inp1, inp2):
        s = self.new()
        self.gates.append(f"out{s} = out{inp1} | out{inp2}")
        return s
    
    def xor_g(self, inp1, inp2):
        s = self.new()
        self.gates.append(f"out{s} = out{inp1} ^ out{inp2}")
        return s
    
    def const(self, val):
        s = self.new()
        self.gates.append(f"out{s} = {val}")
        return s

# Build the circuit
c = Circuit()

# Signals 0-31 are inputs (bits of N, signal 0 = MSB)
for i in range(32):
    c.copy(i, i)
