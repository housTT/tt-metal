#!/usr/bin/env python3
"""
Generate circuit for fib(isqrt(N)) % 2^32
"""

class Circuit:
    def __init__(self):
        self.gates = []
        self.next_sig = 0
        
    def new(self):
        s = self.next_sig
        self.next_sig += 1
        return s
    
    def const(self, out, val):
        self.gates.append(f"out{out} = {val}")
    
    def copy(self, out, inp):
        self.gates.append(f"out{out} = out{inp}")
    
    def not_g(self, out, inp):
        self.gates.append(f"out{out} = ~out{inp}")
    
    def and_g(self, out, inp1, inp2):
        self.gates.append(f"out{out} = out{inp1} & out{inp2}")
    
    def or_g(self, out, inp1, inp2):
        self.gates.append(f"out{out} = out{inp1} | out{inp2}")
    
    def xor_g(self, out, inp1, inp2):
        self.gates.append(f"out{out} = out{inp1} ^ out{inp2}")
    
    def write(self, fname):
        with open(fname, 'w') as f:
            for g in self.gates[:32000]:
                f.write(g + '\n')


# Generate identity gate for signals 0-31 (pass-through)
c = Circuit()

# Signals 0-31 are inputs, pass them through
for i in range(32):
    c.copy(i, i)

c.write("/app/gates.txt")
print(f"Generated {len(c.gates)} gates")
