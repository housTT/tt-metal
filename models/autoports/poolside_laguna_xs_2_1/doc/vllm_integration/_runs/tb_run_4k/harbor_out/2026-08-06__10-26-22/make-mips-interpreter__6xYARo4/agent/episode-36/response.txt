// MIPS ELF Interpreter for Doom
const fs = require('fs');
const path = require('path');

// Memory
class Memory {
  constructor(size = 32 * 1024 * 1024) {
    this.data = Buffer.alloc(size);
  }
  readU32(a) { return this.data.readUInt32LE(a); }
  writeU32(a, v) { this.data.writeUInt32LE(v >>> 0, a); }
  readU8(a) { return this.data[a]; }
  writeU8(a, v) { this.data[a] = v & 0xff; }
  readString(a, max = 256) {
    let s = '';
    for (let i = 0; i < max; i++) {
      const b = this.data[a + i];
      if (b === 0) break;
      s += String.fromCharCode(b);
    }
    return s;
  }
}

// Registers
class Regs {
  constructor() {
    this.r = new Array(32).fill(0);
    this.pc = 0;
  }

// MIPS CPU
class CPU {
  constructor(binaryPath) {
    this.mem = new Memory();
    this.regs = new Regs();
    this.binaryPath = binaryPath;
  }
  
  load() {
    const data = fs.readFileSync(this.binaryPath);
    console.log('Loaded', data.length, 'bytes');
    
    if (data[0] !== 0x7f || data[1] !== 0x45 || data[2] !== 0x4c || data[3] !== 0x46)
      throw new Error('Not ELF');
    
    const entry = data.readUInt32LE(24);
    const phoff = data.readUInt32LE(28);
    const phentSize = data.readUInt16LE(42);
    const phnum = data.readUInt16LE(44);
    
    console.log('Entry: 0x' + entry.toString(16));
    
    let off = phoff;
    for (let i = 0; i < phnum; i++) {
      const type = data.readUInt32LE(off);
      const vaddr = data.readUInt32LE(off + 8);
      const foff = data.readUInt32LE(off + 12);
      const fsize = data.readUInt32LE(off + 16);
      if (type === 1) {
        console.log('Loading to 0x' + vaddr.toString(16));
        this.mem.data.set(data.slice(foff, foff + fsize), vaddr);

      off += phentSize;

  step() {
    const instr = this.mem.readU32(this.regs.pc);
    if (instr === 0) return false;
    
    const op = instr >>> 26;
    const rs = (instr >>> 21) & 31;
    const rt = (instr >>> 16) & 31;
    const imm16 = instr & 0xffff;
    
    // Syscall
    if (op === 0 && (instr & 0x3f) === 0x0c) {
      handleSyscall(this.mem, this.regs);
      this.regs.pc += 4;
      return true;
    }
    
    // JAL
    if (op === 2) {
      this.regs.r[31] = this.regs.pc + 4;
      this.regs.pc = (instr & 0x03ffffff) << 2;
      return true;
    }
    
    // BEQ
    if (op === 4) {
      if (this.regs.r[rs] === this.regs.r[rt]) {
        this.regs.pc += imm16 << 2;
      } else {
        this.regs.pc += 4;
      }
      return true;
    }
    
    // BNE
    if (op === 5) {
      if (this.regs.r[rs] !== this.regs.r[rt]) {
        this.regs.pc += imm16 << 2;
      } else {
        this.regs.pc += 4;
      }
      return true;

    // SLTI
    if (op === 10) {
      this.regs.r[rt] = (this.regs.r[rs] < (imm16 | (imm16 & 0x8000 ? 0x10000 : 0))) ? 1 : 0;
      this.regs.pc += 4;
      return true;
    }
    
    // ADDIU
    if (op === 9) {
      this.regs.r[rt] = (this.regs.r[rs] + imm16) >>> 0;
      this.regs.pc += 4;
      return true;
    }
    
    // LW
    if (op === 35) {
      this.regs.r[rt] = this.mem.readU32(this.regs.r[rs] + imm16);
      this.regs.pc += 4;
      return true;
    }
    
    // SW
    if (op === 43) {
      this.mem.writeU32(this.regs.r[rs] + imm16, this.regs.r[rt]);
      this.regs.pc += 4;
      return true;
    }
    
    // LUI
    if (op === 3) {
      this.regs.r[rt] = imm16 << 16;
      this.regs.pc += 4;
      return true;
    }
    
    // ORI
    if (op === 13) {
      this.regs.r[rt] = this.regs.r[rs] | imm16;
      this.regs.pc += 4;
      return true;
    }
    
    // ADDIU + branch
    if (op === 9) {
      this.regs.r[rt] = (this.regs.r[rs] + imm16) >>> 0;
      this.regs.pc += 4;
      return true;
    }
    
    // SPECIAL instructions
    if (op === 0) {
      const funct = instr & 0x3f;
      switch (funct) {
        case 0x00: this.regs.pc += 4; return true; // SLL
        case 0x02: this.regs.pc += 4; return true; // SRL
        case 0x03: this.regs.pc += 4; return true; // SRA

        case 0x21: this.regs.r[rt] = (this.regs.r[rs] + this.regs.r[rt]) >>> 0; this.regs.pc += 4; return true; // ADDU
        case 0x22: this.regs.r[rt] = (this.regs.r[rs] - this.regs.r[rt]) >>> 0; this.regs.pc += 4; return true; // SUB
        case 0x24: this.regs.r[rt] = this.regs.r[rs] & this.regs.r[rt]; this.regs.pc += 4; return true; // AND
        case 0x25: this.regs.r[rt] = this.regs.r[rs] | this.regs.r[rt]; this.regs.pc += 4; return true; // OR
        case 0x27: this.regs.r[rt] = ~(this.regs.r[rs] | this.regs.r[rt]) >>> 0; this.regs.pc += 4; return true; // NOR

          return true;

    console.log('Unknown opcode ' + op.toString(16) + ' at 0x' + this.regs.pc.toString(16));
