"""Verify retained TT evidence and tokenize saved output only; no model execution."""
import collections
import hashlib
import json
import sys
from pathlib import Path


class NoModelImports:
    def find_spec(self, fullname, path=None, target=None):
        if fullname.split('.')[0] in {'torch', 'ttnn', 'tt_metal', 'models', 'transformers', 'vllm', 'vllm_tt_plugin'}:
            raise RuntimeError('Model/framework import prohibited: ' + fullname)


sys.meta_path.insert(0, NoModelImports())
from tokenizers import Tokenizer

W = Path('/home/hous/dev/tti-release-gemma4')
OUT = Path(__file__).parent
bindings = {}


def read(path):
    path = Path(path)
    raw = path.read_bytes()
    bindings[str(path)] = hashlib.sha256(raw).hexdigest()
    return raw


def load(path):
    return json.loads(read(path))


def checked(ref):
    raw = read(ref['path'])
    assert bindings[ref['path']] == ref['sha256']
    return raw


result_path = W / 'meta_gpqa_expert_both_phase_quality_control_run_v1/result.json'
result = load(result_path)
assert bindings[str(result_path)] == 'c78f72a0ed0e46d7598d8537bd8c7b70a934a2d644c182c6972e243b12658abf'
plan_path = W / 'meta_gpqa_expert_both_phase_quality_control_v1/plan.json'
plan = load(plan_path)

assert owner['returncode'] == 0 and owner['cleanup']['complete']
