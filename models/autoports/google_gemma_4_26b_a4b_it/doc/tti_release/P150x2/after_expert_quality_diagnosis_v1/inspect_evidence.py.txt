bindings = {}


def read(path):
    path = Path(path)
    raw = path.read_bytes()
    bindings[str(path)] = hashlib.sha256(raw).hexdigest()
    return raw


def load(path):
    return json.loads(read(path))


def checked(binding):
    raw = read(binding['path'])
    assert bindings[binding['path']] == binding['sha256'], binding
    return raw


def sample(path, doc_id):
    # JSONL uses LF; splitlines() also splits Unicode separators inside strings.
    return next(row for line in read(path).split(b'\n') if line
                for row in [json.loads(line)] if row['doc_id'] == doc_id)
